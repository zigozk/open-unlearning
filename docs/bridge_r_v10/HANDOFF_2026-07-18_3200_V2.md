From the project parent directory:

```bash
cd /home/zkzhang/unlearn
python3 - <<'PY'
import json, hashlib, pathlib

root = pathlib.Path('/home/zkzhang/unlearn/open-unlearning')
api = root / 'data/atomic_tofu/v2.0-rc1/api/eval_extension'
run = api / 'codex_direct_3200_v4'
inp = api / 'input_units.jsonl'
out = run / 'candidate_outputs_3200_v2.jsonl'
manifest = run / 'run_manifest_3200_v2.json'
report = run / 'reports/run_report_3200_v2.json'
errorq = run / 'reports/error_queue_3200_v2.jsonl'

input_ids = [json.loads(line)['unit_id'] for line in inp.read_text().splitlines() if line.strip()]
records = [json.loads(line) for line in out.read_text().splitlines() if line.strip()]
ids = [r['unit_id'] for r in records]
sha = hashlib.sha256(out.read_bytes()).hexdigest()
m = json.loads(manifest.read_text())
r = json.loads(report.read_text())

bad_shape = []
perturbations = 0
for rec in records:
    if set(rec) != {'unit_id', 'candidate'}:
        bad_shape.append(rec.get('unit_id'))
        continue
    cand = rec['candidate']
    if set(cand) != {'paraphrased_question', 'paraphrased_answer', 'perturbed_answer'}:
        bad_shape.append(rec['unit_id'])
        continue
    pa = cand['perturbed_answer']
    if not isinstance(pa, list) or len(pa) != 5 or len(set(pa)) != 5 or any(not isinstance(x, str) or not x.strip() for x in pa):
        bad_shape.append(rec['unit_id'])
    perturbations += len(pa) if isinstance(pa, list) else 0

print({
    'manifest_status': m.get('status'),
    'next_pending_batch': m.get('next_pending_batch'),
    'completed_records': m.get('completed_records'),
    'remaining_records': m.get('remaining_records'),
    'input_records': len(input_ids),
    'aggregate_lines': len(out.read_text().splitlines()),
    'json_records': len(records),
    'ordered_ids': ids == input_ids,
    'unique_ids': len(set(ids)),
    'perturbations': perturbations,
    'bad_shape': bad_shape[:10],
    'error_queue_bytes': errorq.stat().st_size,
    'aggregate_sha256': sha,
    'manifest_final_sha256': m.get('final_sha256'),
    'report_final_sha256': r.get('final_sha256'),
})
PY
```
